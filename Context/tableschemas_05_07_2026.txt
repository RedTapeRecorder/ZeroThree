create table public.managers (
  id integer generated by default as identity not null,
  email character varying(255) not null,
  password_hash text not null,
  name text not null,
  created_at timestamp without time zone null default CURRENT_TIMESTAMP,
  constraint managers_pkey primary key (id),
  constraint managers_email_key unique (email)
) TABLESPACE pg_default;

create table public.outlet_photos (
  id uuid not null default gen_random_uuid (),
  outlet_id integer not null,
  cloudinary_url text not null,
  cloudinary_id text not null,
  submitted_by text null,
  submission_source text not null,
  status text not null default 'pending'::text,
  is_current boolean not null default false,
  submitted_at timestamp without time zone null default CURRENT_TIMESTAMP,
  reviewed_at timestamp without time zone null,
  reviewed_by text null,
  rejection_reason text null,
  submitted_by_rider_id integer null,
  constraint outlet_photos_pkey primary key (id),
  constraint outlet_photos_submitted_by_rider_id_fkey foreign KEY (submitted_by_rider_id) references riders (id)
) TABLESPACE pg_default;

create index IF not exists outlet_photos_outlet_id_idx on public.outlet_photos using btree (outlet_id) TABLESPACE pg_default;

create index IF not exists outlet_photos_status_idx on public.outlet_photos using btree (status) TABLESPACE pg_default;

create index IF not exists outlet_photos_current_idx on public.outlet_photos using btree (outlet_id) TABLESPACE pg_default
where
  (is_current = true);
